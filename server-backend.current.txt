// Clean single-file backend: sqlite3 if available, else JSON fallback
const express = require('express')
const cors = require('cors')
const bodyParser = require('body-parser')
const speakeasy = require('speakeasy')
const fs = require('fs')
const path = require('path')

let sqlite3 = null
try { sqlite3 = require('sqlite3') } catch (e) { sqlite3 = null }

const app = express()
const port = process.env.PORT || 4000
const DATA_DIR = path.join(__dirname, 'data')
const COMMENTS_JSON = path.join(DATA_DIR, 'comments.json')

app.use(cors({ origin: ['http://localhost:5173', 'http://localhost:8000'] }))
app.use(bodyParser.json())

// In-memory 2FA store for demo
const store = {}

// SQLite initialization
let db = null
function initSqlite(){
  if(!sqlite3) return null
  const sqlite = sqlite3.verbose()
  if(!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR)
  const dbFile = path.join(DATA_DIR, 'comments.db')
  const d = new sqlite.Database(dbFile)
  d.serialize(()=>{
    d.run(`CREATE TABLE IF NOT EXISTS comments (id INTEGER PRIMARY KEY AUTOINCREMENT, datasetId TEXT, user TEXT, text TEXT, time TEXT, avatar TEXT, likes INTEGER DEFAULT 0, liked INTEGER DEFAULT 0)`)
    d.run(`CREATE TABLE IF NOT EXISTS replies (id INTEGER PRIMARY KEY AUTOINCREMENT, commentId INTEGER, user TEXT, text TEXT, time TEXT, avatar TEXT)`)
  })
  return d
}

db = initSqlite()

function readJsonStore(){
  try{ if(fs.existsSync(COMMENTS_JSON)) return JSON.parse(fs.readFileSync(COMMENTS_JSON,'utf8')||'{}') }catch(e){}
  return {}
}
function writeJsonStore(obj){
  try{ if(!fs.existsSync(DATA_DIR)) fs.mkdirSync(DATA_DIR); fs.writeFileSync(COMMENTS_JSON, JSON.stringify(obj, null, 2)) }catch(e){ console.warn('Failed write JSON store', e) }
}

function migrateJsonToSqlite(){
  if(!db) return
  try{
    if(fs.existsSync(COMMENTS_JSON)){
      const raw = fs.readFileSync(COMMENTS_JSON,'utf8')
      const obj = JSON.parse(raw || '{}')
      const insert = db.prepare('INSERT INTO comments (datasetId,user,text,time,avatar,likes,liked) VALUES (?,?,?,?,?,?,?)')
      for(const [datasetId, list] of Object.entries(obj)){
        for(const c of list){
          insert.run(datasetId, c.user||'Unknown', c.text||'', c.time||new Date().toISOString(), c.avatar||'', c.likes||0, c.liked?1:0)
        }
      }
      insert.finalize()
    }
  }catch(e){ console.warn('Migration failed', e) }
}

if(db) migrateJsonToSqlite()

const API_KEY = process.env.API_KEY || 'dev-key'
function requireApiKey(req, res, next){
  const key = req.headers['x-api-key'] || req.query.api_key
  if(key === API_KEY) return next()
  return res.status(401).json({ ok:false, error:'unauthorized' })
}

// 2FA endpoints
app.post('/api/2fa/provision', (req, res) => {
  const { userId = 'demo-user', email = 'user@example.com' } = req.body || {}
  const secret = speakeasy.generateSecret({ length: 20, name: `NexusAnalytics:${email}` })
  store[userId] = { secret: secret.base32 }
  return res.json({ ok: true, secret: secret.base32, otpauth_url: secret.otpauth_url })
})

app.post('/api/2fa/verify', (req, res) => {
  const { userId = 'demo-user', token } = req.body || {}
  const entry = store[userId]
  if (!entry || !entry.secret) return res.status(404).json({ ok: false, error: 'no-secret' })
  const verified = speakeasy.totp.verify({ secret: entry.secret, encoding: 'base32', token, window: 1 })  
  return res.json({ ok: true, verified: !!verified })
})

app.get('/health', (req, res) => res.json({ ok: true }))

// Comments API
app.get('/api/comments/:datasetId', (req, res) => {
  const { datasetId } = req.params
  if(db){
    db.all('SELECT * FROM comments WHERE datasetId = ? ORDER BY id DESC', [datasetId], (err, rows)=>{     
      if(err) return res.status(500).json({ ok:false, error:'db' })
      const ids = rows.map(r=>r.id)
      if(ids.length===0) return res.json({ ok:true, comments: [] })
      db.all(`SELECT * FROM replies WHERE commentId IN (${ids.map(()=>'?').join(',')})`, ids, (er, reps)=>{
        if(er) return res.json({ ok:true, comments: rows.map(r=>({ ...r, replies: [] })) })
        const map = {}
        for(const rp of reps){ map[rp.commentId] = map[rp.commentId]||[]; map[rp.commentId].push(rp) }    
        const out = rows.map(r=> ({ id: r.id, user: r.user, text: r.text, time: r.time, avatar: r.avatar, likes: r.likes, liked: !!r.liked, replies: map[r.id]||[] }))
        return res.json({ ok:true, comments: out })
      })
    })
    return
  }
  const store = readJsonStore()
  const list = store[datasetId] || []
  res.json({ ok:true, comments: list })
})

app.post('/api/comments/:datasetId', requireApiKey, (req, res) => {
  const { datasetId } = req.params
  const { user = 'Anonymous', text = '' } = req.body || {}
  if(!text || !text.trim()) return res.status(400).json({ ok:false, error:'empty' })
  const now = new Date().toISOString()
  if(db){
    db.run('INSERT INTO comments (datasetId,user,text,time,avatar,likes,liked) VALUES (?,?,?,?,?,?,?)', [datasetId, user, text, now, (user||'A').slice(0,2).toUpperCase(), 0, 0], function(err){
      if(err) return res.status(500).json({ ok:false, error:'db' })
      db.get('SELECT * FROM comments WHERE id = ?', [this.lastID], (e, row)=>{
        if(e) return res.status(500).json({ ok:false, error:'db' })
        return res.json({ ok:true, comment: { id: row.id, user: row.user, text: row.text, time: row.time, avatar: row.avatar, likes: row.likes, liked: !!row.liked, replies: [] } })
      })
    })
    return
  }
  const store = readJsonStore()
  const comment = { id: Date.now(), user, text, time: now, avatar: (user||'A').slice(0,2).toUpperCase(), likes:0, liked:false, replies:[] }
  store[datasetId] = [comment, ...(store[datasetId]||[])]
  writeJsonStore(store)
  res.json({ ok:true, comment })
})

app.delete('/api/comments/:datasetId/:commentId', requireApiKey, (req, res) => {
  const { datasetId, commentId } = req.params
  if(db){
    db.run('DELETE FROM comments WHERE id = ? AND datasetId = ?', [Number(commentId), datasetId], function(err){
      if(err) return res.status(500).json({ ok:false, error:'db' })
      db.run('DELETE FROM replies WHERE commentId = ?', [Number(commentId)])
      return res.json({ ok:true })
    })
    return
  }
  const store = readJsonStore(); store[datasetId] = (store[datasetId]||[]).filter(c=>String(c.id)!==String(commentId)); writeJsonStore(store); res.json({ ok:true })
})

      if(err) return res.status(500).json({ ok:false, error:'db' })
      db.get('SELECT * FROM replies WHERE id = ?', [this.lastID], (e, row)=>{

})

app.post('/api/comments/:datasetId/:commentId/toggle-like', requireApiKey, (req, res) => {
  const { datasetId, commentId } = req.params
  if(db){
    db.get('SELECT * FROM comments WHERE id = ? AND datasetId = ?', [Number(commentId), datasetId], (err, row)=>{
      if(err || !row) return res.status(404).json({ ok:false })
      const liked = row.liked?0:1; const likes = liked? (row.likes||0)+1 : Math.max(0,(row.likes||0)-1)
      db.run('UPDATE comments SET liked = ?, likes = ? WHERE id = ?', [liked, likes, Number(commentId)], (e)=>{
        if(e) return res.status(500).json({ ok:false })
        db.get('SELECT * FROM comments WHERE id = ?', [Number(commentId)], (er, updated)=>{ res.json({ ok:true, comment: { ...updated, liked: !!updated.liked } }) })
      })
    })
    return
  }
  const store = readJsonStore(); store[datasetId] = (store[datasetId]||[]).map(c => {
    if(String(c.id)===String(commentId)){ const liked = !c.liked; const likes = liked? (c.likes||0)+1 : Math.max(0,(c.likes||0)-1); return { ...c, liked, likes } } return c
  }); writeJsonStore(store); const updated = (store[datasetId]||[]).find(c=>String(c.id)===String(commentId)); res.json({ ok:true, comment: updated })
})

app.listen(port, () => console.log(`2FA backend listening on http://localhost:${port}`))

// In-memory store for demo purposes. Replace with DB in production.

  // userId: { secret: 'BASE32SECRET' }

// Simple file-backed comments store (demo). Keys: datasetId -> array of comments
const fs = require('fs')
const path = require('path')
const dataDir = path.join(__dirname, 'data')
const commentsFile = path.join(dataDir, 'comments.json')
let commentsStore = {}
try {
  if (!fs.existsSync(dataDir)) fs.mkdirSync(dataDir)
  if (fs.existsSync(commentsFile)){
    commentsStore = JSON.parse(fs.readFileSync(commentsFile, 'utf8') || '{}')
  }
} catch (e) {
  console.warn('Failed to load comments store', e)
  commentsStore = {}
}

function saveComments(){
  try{ fs.writeFileSync(commentsFile, JSON.stringify(commentsStore, null, 2)) }catch(e){ console.warn('Failed to save comments', e) }
}

// Provision a new 2FA secret for a user
app.post('/api/2fa/provision', (req, res) => {
  const { userId = 'demo-user', email = 'user@example.com' } = req.body || {}
  const secret = speakeasy.generateSecret({ length: 20, name: `NexusAnalytics:${email}` })
  store[userId] = { secret: secret.base32 }
  return res.json({ ok: true, secret: secret.base32, otpauth_url: secret.otpauth_url })
})

// Verify a TOTP token for a user
app.post('/api/2fa/verify', (req, res) => {
  const { userId = 'demo-user', token } = req.body || {}
  const entry = store[userId]
  if (!entry || !entry.secret) return res.status(404).json({ ok: false, error: 'no-secret' })

  const verified = speakeasy.totp.verify({ secret: entry.secret, encoding: 'base32', token, window: 1 })
  return res.json({ ok: true, verified: !!verified })
})

app.post('/api/comments/:datasetId', (req, res) => {
  const { datasetId } = req.params
  const { user = 'Anonymous', text = '' } = req.body || {}
  if (!text || !text.trim()) return res.status(400).json({ ok: false, error: 'empty' })
  const comment = { id: Date.now(), user, text, time: 'Just now', avatar: (user||'A').split(' ').map(s=>s[0]).join('').slice(0,2).toUpperCase(), likes:0, liked:false, replies:[] }
  commentsStore[datasetId] = [comment, ...(commentsStore[datasetId]||[])]
  saveComments()
  res.json({ ok: true, comment })
})

app.delete('/api/comments/:datasetId/:commentId', (req, res) => {
  const { datasetId, commentId } = req.params
  const list = commentsStore[datasetId] || []
  const nid = Number(commentId)
  commentsStore[datasetId] = list.filter(c => c.id !== nid)
  saveComments()
  res.json({ ok: true })
})

app.post('/api/comments/:datasetId/:commentId/reply', (req, res) => {
  const { datasetId, commentId } = req.params
  const { user = 'Anonymous', text = '' } = req.body || {}
  if (!text || !text.trim()) return res.status(400).json({ ok: false, error: 'empty' })
  const list = commentsStore[datasetId] || []
  const nid = Number(commentId)
  const reply = { id: Date.now(), user, text, time: 'Just now', avatar: (user||'A').split(' ').map(s=>s[0]).join('').slice(0,2).toUpperCase() }
  commentsStore[datasetId] = list.map(c => c.id === nid ? { ...c, replies: [...(c.replies||[]), reply] } : c)
  saveComments()
  res.json({ ok: true, reply })
})

app.post('/api/comments/:datasetId/:commentId/toggle-like', (req, res) => {
  const { datasetId, commentId } = req.params
  const list = commentsStore[datasetId] || []
  const nid = Number(commentId)
  let updated = null
  commentsStore[datasetId] = list.map(c => {
    if (c.id === nid){
      const liked = !c.liked
      const likes = liked ? (c.likes||0) + 1 : Math.max(0, (c.likes||0) - 1)
      updated = { ...c, liked, likes }
      return updated
    }
    return c
  })
  saveComments()
  res.json({ ok: true, comment: updated })
})
